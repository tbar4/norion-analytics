# Python dependencies for the MAIN Airflow environment, baked in by the
# Dockerfile. dbt is deliberately NOT here — it lives in its own venv, see
# requirements-dbt.txt.
#
# Do NOT list apache-airflow here — the Dockerfile already pins it.
#
# After editing:  docker compose build && docker compose up -d
#
# --- Matching local env with uv -----------------------------------------
# apache/airflow:3.3.0 runs Python 3.13 (verified with `python --version`
# inside the running container — the docs page still says 3.12). To get an
# editor/pytest environment that resolves the same way the image does:
#
#   uv venv --python 3.13
#   uv pip install -r requirements.txt "apache-airflow==3.3.0" \
#     -c https://raw.githubusercontent.com/apache/airflow/constraints-3.3.0/constraints-3.13.txt
#
# That -c constraints file is Airflow's own tested dependency set and is the
# difference between a clean resolve and a very long backtrack.

# Extract + load. dlt is imported directly by DAG code, so it must live in
# the same environment as Airflow.
#
# Extras, and why each is here:
#   postgres   — the warehouse. Every pipeline lands in it.
#   filesystem — pulls s3fs + botocore, which dlt needs to read s3:// from
#                MinIO. Airflow's own S3Hook does NOT need this; only dlt does.
#
# The duckdb extra was dropped 2026-07-28. Nothing had read a DuckDB file since
# the warehouse moved to Postgres — the example DAG and pipeline that used one
# were deleted, and dbt now talks to Postgres over the wire.
dlt[postgres,filesystem]==1.29.1

# Renders each dbt model as its own Airflow task.
astronomer-cosmos==1.15.0

# --- Conjunction screening ----------------------------------------------
# The official Vallado SGP4 propagator. Used by
# include/pipelines/conjunction_screening_pipeline.py, which propagates the
# whole catalogue from CelesTrak OMM element sets.
#
# We drive it through sgp4.api.SatrecArray, which propagates M satellites at N
# times into numpy arrays in one call. The per-satellite Python loop is orders
# of magnitude slower and will not finish an all-on-all screen in a DAG run.
sgp4==2.24

# numpy arrives with sgp4 anyway; pinned here because the screening maths
# depends on it directly rather than incidentally.
numpy==2.3.4

# scipy.spatial.cKDTree does the coarse neighbour query that turns an O(N^2)
# all-pairs screen into something tractable, and scipy.special supplies the
# error functions for the 2D Pc integral.
scipy==1.16.3

# NOTE: the old "must match requirements-dbt.txt exactly" rule is GONE, and so
# is the `duckdb==1.5.5` pin it applied to (removed 2026-07-28). That rule
# existed because dlt wrote a DuckDB file which dbt then read from a *different*
# virtualenv, so an on-disk format skew between the two broke the platform.
# Both now talk to a Postgres server over the wire, sharing no file, so there is
# nothing left to keep in sync between these two requirements files.
